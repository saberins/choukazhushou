function spin(){panel.innerHTML=`${head('正在为你抽取…','让好运气转一圈，马上揭晓。','✦ GOOD LUCK')}<div class="result-card" style="padding:44px 20px"><div class="result-emoji" style="animation:spin 1s ease-in-out infinite">🎲</div><div class="result-label">正在抽取</div><h2>好运马上抵达</h2><p>从 ${selected.length} 个选项中随机选择</p></div>`;let style=document.createElement('style');style.textContent='@keyframes spin{0%{transform:rotate(0) scale(.9)}50%{transform:rotate(180deg) scale(1.15)}100%{transform:rotate(360deg) scale(.9)}}';document.head.appendChild(style);setTimeout(()=>{drawn=selected[Math.floor(Math.random()*selected.length)];setPage('result')},1100)}
function result(){const x=current();panel.innerHTML=`${head('抽取结果','这就是今天的幸运选项。','抽中啦！')}<div class="result-card"><div class="result-emoji">${foodIcon(drawn,x.category)}</div><div class="result-label">抽 中 了 ！</div><h2>${esc(drawn)}</h2><p>来自：${esc(x.name)}</p></div><div class="result-buttons"><button class="btn primary" id="accept">就选这个 ✓</button><button class="btn secondary" id="again">再抽一次</button><button class="btn secondary" id="adjust">调整选项</button></div>`;document.querySelector('#accept').onclick=()=>{x.last=drawn;previous=x.id;persist();setPage('confirmed')};document.querySelector('#again').onclick=()=>spin();document.querySelector('#adjust').onclick=()=>setPage('prepare')}
function confirmed(){const x=current();panel.innerHTML=`${head('选择已确认','今天的纠结到此结束。','DONE')}<div class="success"><div class="success-icon">✓</div><h2>今天选 ${esc(drawn)}</h2><p>按自己的节奏，开始今天的${esc(x.name.replace(/吃什么|做什么|看什么/g,''))}吧。</p><div class="confirmed"><span class="cat-icon">${x.icon}</span><span><b>${esc(x.name)}</b><small>已确认：${esc(drawn)}</small></span><span class="badge" style="margin-left:auto">已保存</span></div><div class="success-note">✨ 清单已保存，下次可以直接复用，也可以重新抽取。</div><div class="actions"><button class="btn primary" id="home">返回我的清单</button><button class="text-btn" id="again">再抽一次</button></div></div>`;document.querySelector('#home').onclick=()=>setPage('home');document.querySelector('#again').onclick=()=>{selected=[...x.items];setPage('prepare')}}
function foodIcon(name,category){if(name.includes('包子'))return '🥟';if(name.includes('面'))return '🍜';if(name.includes('三明治'))return '🥪';if(name.includes('咖啡'))return '☕';if(name.includes('电影')||name.includes('剧'))return '🎬';return category.startsWith('🍴')?'🍽️':category.startsWith('🎬')?'🎥':category.startsWith('🗓️')?'🌿':'✨'}
function esc(s){return String(s||'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
